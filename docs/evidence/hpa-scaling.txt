HPA manifest under test
==============================================================
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: civicpulse-backend
  namespace: civicpulse
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: civicpulse-backend
  minReplicas: 2
  maxReplicas: 10
  metrics:
    - type: Resource
      resource:
        name: cpu
        target:
          type: Utilization
          averageUtilization: 60
  behavior:
    scaleDown:
      stabilizationWindowSeconds: 300
    scaleUp:
      stabilizationWindowSeconds: 0
